def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def main():

    stamp=datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%SZ');archive=BOOT/('ordered_startup_failure_'+stamp+'.tar.gz')

    if r.returncode:
        archive.with_suffix('.failure.json').write_text(json.dumps(dict(exit_code=r.returncode,stderr=r.stderr.decode(errors='replace')),indent=2)+'\n')
        raise RuntimeError('Failure evidence transport failed; partial preserved')
    DEST.mkdir(parents=True);seen=set()
    with tarfile.open(archive,'r:gz') as tar:
        for member in tar:
            target=DEST/member.name;target.resolve().relative_to(DEST.resolve())
            if not member.isfile() or member.name in seen:raise ValueError('Unsafe member')
            seen.add(member.name);target.parent.mkdir(parents=True,exist_ok=True);target.write_bytes(tar.extractfile(member).read())
    index=json.loads((DEST/'SERVER_ARCHIVE_INDEX.json').read_text())
    for row in index['files']:assert sha(DEST/row['path'])==row['sha256'] and (DEST/row['path']).stat().st_size==row['bytes']
