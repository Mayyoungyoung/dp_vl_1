def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda:f.read(2**20),b''):h.update(block)
    return h.hexdigest()


def read(path):return json.loads(Path(path).read_text(encoding='utf-8'))


def write(path,value):
    path=Path(path);temp=path.with_name(path.name+'.writing')
    temp.write_text(json.dumps(value,indent=2,allow_nan=False)+'\n',encoding='utf-8');temp.replace(path)


def verify_source():

    m=read(MANIFEST)

    for name,item in m['files'].items():
        path=SOURCE/name;path.resolve().relative_to(SOURCE.resolve())
        if path.is_symlink() or not path.is_file() or path.stat().st_size!=item['bytes'] or sha(path)!=item['sha256']:
            raise ValueError('Frozen actual archive bytes changed: '+name)
    return m


def validate_junit(path,manifest):
    tree=ET.parse(path);suites=list(tree.iter('testsuite'));cases=list(tree.iter('testcase'))
    identities=[[c.get('classname'),c.get('name')] for c in cases]
    if (len(cases)!=20 or len(set(map(tuple,identities)))!=20 or
            sum(int(s.get('tests','0')) for s in suites)!=20 or
            any(int(s.get(k,'0')) for s in suites for k in ('skipped','failures','errors')) or
            any(c.find(k) is not None for c in cases for k in ('skipped','failure','error')) or
            sorted(identities)!=sorted(manifest['expected_pytest_cases'])):
        raise ValueError('All exact20 identities, including17 actual Torch cases, must pass without skips')
    return dict(passed=20,skipped=0,failed=0,errors=0,testcases=identities)

class Journal:
    """One issued record per operation, written and fsynced before that work."""
    def __init__(self,path):
        self.path=Path(path);self.file=self.path.open('x',encoding='utf-8')
        self.issued=[];self.completed=[];self.active=None;self.observed=set()
    def emit(self,event):
        kind=event['event'];key=event['call_id']
        if kind=='issued':
            if self.active is not None or len(self.issued)>=len(EXPECTED_IDS) or key!=EXPECTED_IDS[len(self.issued)]:
                raise ValueError('Duplicate/out-of-order/excess synthetic call refused before work')
        elif kind in ('observed','completed'):
            if key!=self.active:raise ValueError('Synthetic completion without exact issued operation')
            if kind=='observed' and key in self.observed:raise ValueError('Duplicate observed synthetic operation')
            if kind=='completed' and key!='reference_cache' and key not in self.observed:
                raise ValueError('Loss completed without finite-result observation')
        else:raise ValueError('Unknown progress event')
        self.file.write(json.dumps(dict(event,monotonic_seconds=time.perf_counter()),allow_nan=False)+'\n')
        self.file.flush();os.fsync(self.file.fileno())
        if kind=='issued':self.issued.append(key);self.active=key
        elif kind=='observed':self.observed.add(key)
        else:self.completed.append(key);self.active=None
    def audit(self):
        return dict(issued_ids=self.issued,completed_ids=self.completed,
            issued_loss_calls=sum(x!='reference_cache' for x in self.issued),
            completed_loss_calls=sum(x!='reference_cache' for x in self.completed),
            issued_reference_cache_calls=int('reference_cache' in self.issued),
            completed_reference_cache_calls=int('reference_cache' in self.completed),
            issued_without_certified_completion=[x for x in self.issued if x not in self.completed],
            conservative_budget_policy='Every issued call counts, even if incomplete or nonfinite; no replay',
            model_forwards=0,optimizer_updates=0,real_dataset_reads=0,checkpoint_reads=0)
    def close(self):self.file.close()


def validate_benchmark(q,journal):
    expected=dict(batch_size=32,candidates=4,references_max=9,horizon=24,observed_points=12544,
        dtype='float32',warmup_per_arm=2,measured_repeats_per_arm=3,synthetic_only=True)
    if any(q['settings'].get(k)!=v for k,v in expected.items()):raise ValueError('Full actual-size synthetic setting changed')
    if (set(q['arms'])!=set(ARMS) or q['model_forwards']!=0 or q['optimizer_updates']!=0 or
            q['issued_loss_calls']!=15 or q['completed_loss_calls']!=15 or
            journal.issued!=EXPECTED_IDS or journal.completed!=EXPECTED_IDS):
        raise ValueError('Incomplete/excess synthetic budget; no training follows')
    for arm,record in q['arms'].items():
        calls=record['all_calls']
        if (len(calls)!=5 or [c['call_id'] for c in calls]!=[arm+'/'+str(i) for i in range(5)] or
                [c['phase'] for c in calls]!=['warmup']*2+['measured']*3 or
                len(record['seconds'])!=3 or len(record['warmup_seconds'])!=2 or record['loss_forward_backward_calls']!=5):
            raise ValueError('Missing warmup/measured calls')
        for c in calls:
            if (not c['finite_loss_and_gradients'] or not math.isfinite(c['loss']) or
                    not math.isfinite(c['xyz_gradient_maxabs']) or not math.isfinite(c['seconds']) or c['seconds']<=0 or
                    c['valid_pair_negative_count']<0):raise ValueError('Nonfinite/invalid synthetic result')
        if record['seconds']!=[c['seconds'] for c in calls[2:]]:raise ValueError('Measured seconds inconsistent')
        if record['warmup_seconds']!=[c['seconds'] for c in calls[:2]]:raise ValueError('Warmup seconds inconsistent')
        if record['projected_3000_loss_only_seconds']!=record['median_seconds']*3000:
            raise ValueError('Incorrect loss-only projection')
    return dict(synthetic_loss_calls=15,reference_cache_calls=1,real_dataset_reads=0,checkpoint_reads=0,
        model_forwards=0,optimizer_updates=0,negative_costs_are_reported_not_rejected_or_clamped=True,
        training_auto_launch=False)
