def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def read(path):return json.loads(Path(path).read_text(encoding='utf8'))
def write(path,value):
    path=Path(path);path.parent.mkdir(parents=True,exist_ok=True)
    path.write_text(json.dumps(value,indent=2,ensure_ascii=False,allow_nan=False)+'\n',encoding='utf8')
def copy(source,dest):
    dest=Path(dest);dest.parent.mkdir(parents=True,exist_ok=True);dest.write_bytes(Path(source).read_bytes())
    assert sha(source)==sha(dest)
def index(folder,**metadata):
    files=[dict(path=p.relative_to(folder).as_posix(),bytes=p.stat().st_size,sha256=sha(p))
           for p in sorted(folder.rglob('*')) if p.is_file() and p.name!='LOCAL_ARTIFACT_INDEX.json']
    write(folder/'LOCAL_ARTIFACT_INDEX.json',dict(files=files,local_file_count=len(files),**metadata))
    for item in files:assert sha(folder/item['path'])==item['sha256']
    return dict(folder=folder.as_posix(),files=len(files),index_sha256=sha(folder/'LOCAL_ARTIFACT_INDEX.json'))
