def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda:f.read(2**20),b''):h.update(chunk)
    return h.hexdigest()
def read(path):return json.loads(Path(path).read_text(encoding='utf8'))
def write(path,value):

def digest(raw):return hashlib.sha256(raw).hexdigest()
def frozen(path,raw):
    if path.exists():
        if path.read_bytes()!=raw:raise FileExistsError('Refuse to replace frozen bytes: '+str(path))
    else:
        with path.open('xb') as f:f.write(raw)

def synthetic_tests(runner):
    scope={'__name__':'local_recipe_fixture'};exec(compile(runner,'<frozen_recipe>','exec'),scope)
    def xml(kind):
        root=ET.Element('testsuites');suite=ET.SubElement(root,'testsuite',tests='48',errors='0',failures='0',skipped='0')
        for index,(group,name) in enumerate(scope['EXPECTED_CASES']):
            if kind=='missing' and index==0:continue
            case=ET.SubElement(suite,'testcase',classname=group,name=name)
            if index==0:
                if kind in ('error','failure','skipped'):ET.SubElement(case,kind)
                if kind=='substitution':case.set('name','unrelated_test')
        if kind=='duplicate':ET.SubElement(suite,'testcase',classname=group,name=name)
        if kind=='count':suite.set('tests','49')
        return ET.tostring(root)
    with tempfile.TemporaryDirectory() as folder:
        path=Path(folder)/'junit.xml';path.write_bytes(xml('valid'))
        assert scope['validate_junit'](path)['tests']==48
        for kind in ('missing','error','failure','skipped','substitution','duplicate','count'):
            path.write_bytes(xml(kind))
            try:scope['validate_junit'](path)
            except ValueError:pass
            else:raise AssertionError('Accepted '+kind)
