def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda:f.read(2**20),b''):h.update(chunk)
    return h.hexdigest()
def read(path):return json.loads(Path(path).read_text(encoding='utf8'))
def write(path,value):
    path=Path(path);temporary=path.with_name(path.name+'.writing')
    temporary.write_text(json.dumps(value,indent=2,allow_nan=False)+'\n',encoding='utf8');temporary.replace(path)

def validate_junit(path):

    if not suites or any(int(s.get(k,'0')) for s in suites for k in ('errors','failures','skipped')):
        raise ValueError('Require all39 actual tests without errors/failures/skips')
    for case in root.iter('testcase'):
        if any(case.find(k)is not None for k in ('error','failure','skipped')):raise ValueError('Every testcase must pass')
        actual.append((case.get('classname'),case.get('name')))
    if (len(actual)!=39 or len(set(actual))!=39 or sorted(actual)!=sorted(EXPECTED_CASES)
            or sum(int(s.get('tests','0')) for s in suites)!=39):

    return dict(tests=39,passed=39,skipped=0,transport=20,real_CPU_Torch=3,old_probe=19,
                cuda_or_official_model_calls=0,actual_testcases=actual)

        if file.is_symlink() or not file.is_file() or file.stat().st_size!=item['bytes'] or sha(file)!=item['sha256']:
            raise ValueError('Frozen Git archive bytes changed: '+name)
    if sha(SOURCE/'scripts/probe_hamster3d_train6.py')!=OLD_SHA:
        raise ValueError('Original b127 actually executed source bytes changed')
    return dict(path=str(MANIFEST),sha256=MANIFEST_SHA,code_commit=COMMIT,files=len(value['files']),
                archive_sha256=value['archive_sha256'],original_b127_source_sha256=OLD_SHA)
