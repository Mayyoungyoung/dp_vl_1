def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def read(path):return json.loads(Path(path).read_text(encoding='utf8'))
def write(path,value):
    path=Path(path);temporary=path.with_name(path.name+'.writing')
    temporary.write_text(json.dumps(value,indent=2,allow_nan=False)+'\n',encoding='utf8');temporary.replace(path)
def canonical(name):return re.sub(r'[-_.]+','-',name).lower()

def inventory():
    values={}
    for dist in metadata.distributions():
        name=canonical(dist.metadata['Name']);location=Path(dist.locate_file('')).resolve();location.relative_to(ENV.resolve())
        if name in values:raise ValueError('Duplicate package metadata: '+name)
        values[name]=dist.version
    return dict(sorted(values.items()))

def dependency_check(manifest,before):
    from packaging.requirements import Requirement
    from packaging.specifiers import SpecifierSet
    from packaging.markers import default_environment
    if before!=manifest['original_packages']:raise ValueError('Existing package population differs from sealed original environment')
    if set(manifest['packages'])&set(before):raise ValueError('Fresh supplement cannot replace an existing package')
    planned={**before,**{name:item['version'] for name,item in manifest['packages'].items()}};checks=[]
    environment=default_environment();environment['extra']=''
    if environment['sys_platform']!='linux' or environment['python_version']!='3.11':raise ValueError('Linux/Python3.11 only')
    for name,item in manifest['packages'].items():
        if environment['python_full_version'] not in SpecifierSet(item['requires_python']):raise ValueError('Incompatible Python '+name)
        for text in item['requires_dist']:
            req=Requirement(text)
            if req.marker and not req.marker.evaluate(environment):continue
            target=canonical(req.name)
            if target not in planned or planned[target] not in req.specifier:raise ValueError('Unsatisfied planned dependency: '+text)
            checks.append(dict(requirement=text,selected_version=planned[target],preexisting=target in before))
    return checks

def unchanged(before,after,manifest):
    if any(after.get(name)!=version for name,version in before.items()):raise ValueError('An original package was changed/removed')
    added={name:version for name,version in after.items() if name not in before}
    expected={name:item['version'] for name,item in manifest['packages'].items()}
    if added!=expected:raise ValueError('Only the four fixed new test dependencies may be added')
    return added

def accelerate_hashes():

def command(name,args,timeout=60):
    file=OUT/(name+'.txt')
    with file.open('wb') as stream:
        result=subprocess.run(args,stdout=stream,stderr=subprocess.STDOUT,timeout=timeout)
    write(OUT/(name+'.command.json'),dict(command=args,exit_code=result.returncode,output_sha256=sha(file)))
    if result.returncode:raise RuntimeError('Recorded command failed: '+name)

def original_files(manifest):
    marker=ORIGINAL/'environment.completed.json';value=read(marker);folder=Path(value['session']);folder.resolve().relative_to(ORIGINAL.resolve())
    state=read(folder/'status.json')
    if (state.get('status')!='completed' or state.get('exit_code')!=0 or sha(folder/'receipt.json')!=value['receipt_sha256']
            or sha(folder/'pip_freeze.txt')!=manifest['original_pip_freeze_sha256']):
        raise ValueError('Actual original environment identity differs')
    paths=[ORIGINAL/'preparation_identity.json',marker]
    paths += [path for path in folder.rglob('*') if path.is_file()]
    return {str(path):sha(path) for path in paths}
