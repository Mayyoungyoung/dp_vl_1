def differing(a,b,prefix=''):
    import torch
    import numpy as np
    if type(a)!=type(b):return [dict(path=prefix,reason='type')]
    if isinstance(a,dict):
        if set(a)!=set(b):return [dict(path=prefix,reason='keys')]
        return [r for k in a for r in differing(a[k],b[k],prefix+'/'+str(k))]
    if isinstance(a,(list,tuple)):
        if len(a)!=len(b):return [dict(path=prefix,reason='length')]
        return [r for i in range(len(a)) for r in differing(a[i],b[i],prefix+'/'+str(i))]
    if torch.is_tensor(a):
        if torch.equal(a,b):return []
        return [dict(path=prefix,reason='tensor',shape=list(a.shape),dtype=str(a.dtype),
                     max_abs=float((a.double()-b.double()).abs().max()),changed_elements=int((a!=b).sum()))]
    if isinstance(a,np.ndarray):
        return [] if np.array_equal(a,b) else [dict(path=prefix,reason='numpy')]
    return [] if a==b else [dict(path=prefix,reason='scalar',left=a,right=b)]
