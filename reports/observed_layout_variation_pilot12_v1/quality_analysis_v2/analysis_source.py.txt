sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
read=lambda p:json.loads(Path(p).read_text(encoding='utf8'))
def write(path,value):Path(path).write_text(json.dumps(value,indent=2,allow_nan=False)+'\n',encoding='utf8')
def lines(path):return [json.loads(x) for x in Path(path).read_text(encoding='utf8').splitlines() if x]
def local(name):
    options=[p for p in (REPORT/name,BINARY/name) if p.is_file()]
    if len(options)!=1:raise ValueError('Exactly one unchanged original required: '+name)
    return options[0]
def normalize(value):return json.loads(json.dumps(value))
def describe(values):
    x=np.asarray(values,dtype=float)
    return dict(n=len(x),min=float(x.min()),median=float(np.median(x)),mean=float(x.mean()),p95=float(np.quantile(x,.95)),max=float(x.max())) if len(x) else dict(n=0)

def frozen_import():
    manifest=read(REPORT/'recipe/layout_variation_sources_eba0994.json')
    if manifest['code_commit']!=REV:raise ValueError('Wrong frozen source')
    archive=ROOT/'.bootstrap/layout_variation_eba0994.tar.gz'
    if sha(archive)!=manifest['archive_sha256']:raise ValueError('Wrong Git archive')
    if not FROZEN.exists():
        FROZEN.mkdir()
        with tarfile.open(archive) as tar:
            for m in tar.getmembers():
                if m.isdir():continue
                if not m.isfile() or Path(m.name).is_absolute() or '..' in Path(m.name).parts:raise ValueError('Unsafe archive entry')
                raw=tar.extractfile(m).read();expected=manifest['files'][m.name]
                if hashlib.sha256(raw).hexdigest()!=expected['sha256'] or len(raw)!=expected['bytes']:raise ValueError('Archive source mismatch')
                dest=FROZEN/m.name;dest.parent.mkdir(parents=True,exist_ok=True);dest.write_bytes(raw)
    for name,item in manifest['files'].items():
        if sha(FROZEN/name)!=item['sha256']:raise ValueError('Frozen local source changed: '+name)
    sys.path.insert(0,str(FROZEN));sys.dont_write_bytecode=True
    from scripts import collect_observed_layout_variation as collector
    return collector,manifest

def plot_target(parent,target,rows,config,geometry,rgb,path):
    import matplotlib;matplotlib.use('Agg')
    import matplotlib.pyplot as plt
    from matplotlib.patches import Rectangle
    fig,axes=plt.subplots(3,6,figsize=(19,10),squeeze=False)
    allpts=np.concatenate([r['_xyz'] for r in rows if len(r['_xyz'])]);goal=geometry['target_centers'][target]
    limits=[(min(-.1,float(allpts[:,0].min())-.04),max(.6,float(allpts[:,0].max())+.04)),
            (min(-.3,float(allpts[:,1].min())-.04),max(.3,float(allpts[:,1].max())+.04)),
            (.70,max(1.05,float(allpts[:,2].max())+.04))]
    for idx,row in enumerate(rows):
        for view,(dims,label) in enumerate((((0,1),'XY'),((0,2),'XZ'))):
            ax=axes[idx//3,(idx%3)*2+view];a,b=dims;color='#087f5b' if row['accepted'] else '#b42318'
            for c,h in zip(geometry['obstacle_centers'],geometry['obstacle_halfsizes']):
                ax.add_patch(Rectangle((c[a]-h[a],c[b]-h[b]),2*h[a],2*h[b],fc='#888888',alpha=.5))
                ax.add_patch(Rectangle((c[a]-h[a]-.02,c[b]-h[b]-.02),2*h[a]+.04,2*h[b]+.04,fc='none',ec='#555555',ls=':',lw=.7))
            xyz=row['_xyz'];h24=row['_h24']
            ax.plot(xyz[:,a],xyz[:,b],color=color,lw=.8,alpha=.8)
            ax.plot(h24[:,a],h24[:,b],color='#2b4eff',ls='--',lw=.65,alpha=.7)
            ax.scatter([xyz[0,a]],[xyz[0,b]],c='black',s=8);ax.scatter([goal[a]],[goal[b]],c='#d6a700',marker='*',s=35)
            ax.set_xlim(limits[a]);ax.set_ylim(limits[b]);ax.grid(alpha=.18)
            if view==0:
                kind='unknown' if row['raw_type'] is None else '/'.join(row['raw_type'])
                ax.set_title(f"slot{idx} {'ACCEPT' if row['accepted'] else row['failure_category']}\n{kind}; {row['raw_length_m']:.2f}m",fontsize=8)
            else:ax.set_title(label+' raw; blue H24',fontsize=8)
            ax.tick_params(labelsize=6)
    fig.suptitle(f'{parent} target{target} | all nine actual proposals | solid green accepted/red failed; blue H24\n'
                 'Filled boxes true cuboids; dotted 2cm projection; label-only QA, no route changes',fontsize=12)
    fig.tight_layout(rect=(0,0,1,.94));fig.savefig(path,dpi=120);plt.close(fig)
