def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda:f.read(2**20),b''):h.update(block)
    return h.hexdigest()


def read(path):return json.loads(Path(path).read_text(encoding='utf-8'))
def write(path,value):
    path=Path(path);temp=path.with_name(path.name+'.writing')
    temp.write_text(json.dumps(value,indent=2,allow_nan=False)+'\n',encoding='utf-8');temp.replace(path)


def verify_source():
    if sha(SOURCE_MANIFEST)!=MANIFEST_SHA:raise ValueError('Frozen source manifest changed')
    manifest=read(SOURCE_MANIFEST)
    if manifest['code_commit']!=COMMIT or manifest['expected_tests']!=114:raise ValueError('Wrong source/test manifest')
    for name,item in manifest['files'].items():
        file=SOURCE/name
        if file.is_symlink() or not file.is_file() or file.stat().st_size!=item['bytes'] or sha(file)!=item['sha256']:
            raise ValueError('Actual archive bytes differ: '+name)
    return manifest


def validate_junit(path,manifest):
    tree=ET.parse(path).getroot();suites=list(tree.iter('testsuite'));actual=[]
    if not suites or any(int(s.get(k,'0')) for s in suites for k in ('skipped','failures','errors')):
        raise ValueError('Every actual test must pass with zero skips')
    for case in tree.iter('testcase'):
        if any(case.find(k) is not None for k in ('skipped','failure','error')):raise ValueError('Nonpassing testcase')
        actual.append([case.get('classname'),case.get('name')])
    if (len(actual)!=114 or len({tuple(v) for v in actual})!=114 or
            sum(int(s.get('tests','0')) for s in suites)!=114 or
            sorted(actual)!=sorted(manifest['expected_pytest_cases'])):
        raise ValueError('Require exact21 new and93 old test identities, without substitutes')
    return dict(tests=114,passed=114,skipped=0,errors=0,failures=0,actual_testcases=actual)


def prior_stage(stage,manifest):
    state=read(RUN/(stage+'.status.json'));identity_path=RUN/(stage+'_identity.json')
    receipt_path=RUN/(stage+'_receipt.json');identity=read(identity_path);receipt=read(receipt_path)
    if (state.get('status')!='completed' or state.get('exit_code')!=0 or state.get('code_commit')!=COMMIT
            or receipt.get('status')!='completed' or receipt.get('child_exit_code')!=0
            or receipt.get('identity_sha256')!=sha(identity_path)
            or identity.get('code_commit')!=COMMIT or identity.get('stage')!=stage
            or identity.get('runner_sha256')!=sha(__file__) or identity.get('source_manifest_sha256')!=MANIFEST_SHA):
        raise ValueError('Exact recorded earlier stage must be complete: '+stage)
    for name,digest in receipt['sealed_artifacts'].items():
        file=RUN/name
        file.resolve().relative_to(RUN.resolve())
        if sha(file)!=digest:raise ValueError('Previous receipt artifact changed')
    if stage=='tests':validate_junit(RUN/'validation/pytest.xml',manifest)
    if stage=='prepare':
        for name,digest in receipt['corpus_artifacts'].items():
            file=CORPUS/name;file.resolve().relative_to(CORPUS.resolve())
            if sha(file)!=digest:raise ValueError('Prepared corpus identity changed')
    return dict(status_sha256=sha(RUN/(stage+'.status.json')),identity_sha256=sha(identity_path),receipt_sha256=sha(receipt_path))


def no_active_duplicate():
    hits=[]
    for folder in Path('/proc').iterdir():
        if not folder.name.isdigit() or int(folder.name)==os.getpid():continue
        try:tokens=(folder/'cmdline').read_bytes().replace(b'\0',b' ').decode(errors='replace')
        except (OSError,ProcessLookupError):continue
        if 'collect_observed_layout_variation' in tokens and str(CORPUS) in tokens:hits.append(int(folder.name))
    if hits:raise RuntimeError('Existing variable-layout process PIDs: '+str(hits))
    return dict(matching_active_pids=hits,other_process_payload_logged=False)


def resources():
    from scripts.collect_two_row_extension import directory_bytes
    usage=dict(corpus_bytes=directory_bytes(CORPUS),run_bytes=directory_bytes(RUN))
    free=shutil.disk_usage(P).free
    mem={line.split(':')[0]:int(line.split()[1])*1024 for line in Path('/proc/meminfo').read_text().splitlines() if ':' in line}
    available=mem.get('MemAvailable',0)
    if sum(usage.values())+RESERVE>LIMIT or free<RESERVE or available<RESERVE:
        raise RuntimeError('New-corpus8GiB /512MiB disk and available-RAM reserve gate failed')
    return dict(**usage,internal_logical_budget_bytes=LIMIT,reserve_bytes=RESERVE,filesystem_free_bytes=free,
                host_memory_available_bytes=available,measurement='metadata only; no raw outcomes or images',
                memory_scope='available-RAM preflight, not an RLIMIT_AS or total-host allocation claim')


@contextmanager
def own_xvfb(out):
    """Start and clean up only this directly-owned display server."""
    xvfb=P/'.observation-deps/xorg/usr/bin/Xvfb';sim=P/'.observation-deps/CoppeliaSim'
    if not xvfb.is_file() or not sim.is_dir() or not (P/'.venv-sim/bin/python').is_file():raise ValueError('Existing private simulator environment absent')
    xdg=out/'xdg';xdg.mkdir();xdg.chmod(0o700)
    env=os.environ.copy();env.update(COPPELIASIM_ROOT=str(sim),LD_LIBRARY_PATH=str(sim)+':'+env.get('LD_LIBRARY_PATH',''),
        QT_QPA_PLATFORM_PLUGIN_PATH=str(sim),LIBGL_ALWAYS_SOFTWARE='1',MESA_LOADER_DRIVER_OVERRIDE='llvmpipe',
        LP_NUM_THREADS='1',XDG_RUNTIME_DIR=str(xdg))
    proc=None;info=dict(xvfb_binary=str(xvfb),xvfb_sha256=sha(xvfb),cleanup_scope='only own Popen child')
    with (out/'displaynum').open('w') as display,(out/'xvfb.log').open('wb') as log:
        try:
            proc=subprocess.Popen([str(xvfb),'-displayfd',str(display.fileno()),'-screen','0','640x480x24','-nolisten','tcp'],
                pass_fds=(display.fileno(),),stdout=log,stderr=subprocess.STDOUT,env=env)
            info['pid']=proc.pid;info['parent_pid']=os.getpid();write(out/'xvfb_receipt.json',info)
            for _ in range(40):
                if proc.poll() is not None:raise RuntimeError('Own Xvfb exited during startup')
                if (out/'displaynum').stat().st_size:break
                time.sleep(.25)
            number=(out/'displaynum').read_text().strip()
            if not number.isdigit():raise RuntimeError('Own displayfd did not return a display')
            env['DISPLAY']=':'+number;info['display']=env['DISPLAY'];write(out/'xvfb_receipt.json',info)
            yield env
        finally:
            if proc is not None:
                if proc.poll() is None:
                    proc.terminate()
                    try:proc.wait(timeout=5)
                    except subprocess.TimeoutExpired:proc.kill();proc.wait(timeout=5)
                info['exit_code']=proc.returncode;info['cleanup_complete']=True
                write(out/'xvfb_receipt.json',info)
