    command='/home/wzy/dpvlm/route_set_v1/.venv/bin/python -c '+shlex.quote("import base64;exec(base64.b64decode('"+encoded+"'))")
    return subprocess.run(['ssh','wzy3090',command],check=True,text=True,capture_output=True).stdout
