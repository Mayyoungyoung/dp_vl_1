def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda:f.read(2**20),b''):h.update(chunk)
    return h.hexdigest()

def read(path):return json.loads(Path(path).read_text(encoding='utf-8'))
def write(path,value):
    path=Path(path);temp=path.with_suffix(path.suffix+'.writing')
    temp.write_text(json.dumps(value,indent=2,allow_nan=False)+'\n',encoding='utf-8');temp.replace(path)

        raise ValueError('Require zero actual errors/failures/skips')
    cases=[]
    for case in xml.iter('testcase'):
        if any(case.find(k) is not None for k in ('error','failure','skipped')):
            raise ValueError('Every exact testcase must pass')
        cases.append((case.get('classname'),case.get('name')))
    if len(cases)!=32 or len(set(cases))!=32 or sorted(cases)!=sorted(EXPECTED_CASES):
        raise ValueError('Require exact 13 diffusion-analysis and 19 probe tests; no substitution')
    return dict(tests=32,passed=32,skipped=0,errors=0,failures=0,pure_tests=32,
                diffusion_analysis=13,hamster_probe=19,actual_testcases=cases)

def verify_source():

    for name,item in value['files'].items():

            raise ValueError('Immutable source bytes changed: '+name)
    return dict(path=str(manifest_path),sha256=SOURCE_MANIFEST_SHA,files=len(value['files']),
                archive_sha256=value['archive_sha256'])

def digest(value):return hashlib.sha256(value).hexdigest()
def frozen(path,value):
    if path.exists():
        if path.read_bytes()!=value:raise FileExistsError('Never overwrite different frozen bytes: '+str(path))
    else:
        with path.open('xb') as f:f.write(value)

def self_test(runner):
    namespace={'__name__':'recipe_fixture'};exec(compile(runner,'<recipe>','exec'),namespace)
    def xml(kind):
        root=ET.Element('testsuites');suite=ET.SubElement(root,'testsuite',tests='32',errors='0',failures='0',skipped='0')
        for i,(group,name) in enumerate(namespace['EXPECTED_CASES']):
            if kind=='missing' and i==0:continue
            case=ET.SubElement(suite,'testcase',classname=group,name=name)
            if i==0:
                if kind in ('failure','skipped'):ET.SubElement(case,kind)
                if kind=='renamed':case.set('name','substituted')
        if kind=='duplicate':ET.SubElement(suite,'testcase',classname=group,name=name)
        return ET.tostring(root)
    with tempfile.TemporaryDirectory() as folder:
        path=Path(folder)/'pytest.xml';path.write_bytes(xml('valid'))
        assert namespace['validate_junit'](path)['tests']==32
        for kind in ('missing','failure','skipped','renamed','duplicate'):
            path.write_bytes(xml(kind))
            try:namespace['validate_junit'](path)
            except ValueError:pass
            else:raise AssertionError('Accepted '+kind)
