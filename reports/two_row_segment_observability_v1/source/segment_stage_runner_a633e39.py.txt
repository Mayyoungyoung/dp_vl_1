def sha(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda:f.read(2**20), b''): h.update(block)
    return h.hexdigest()

def read(path): return json.loads(Path(path).read_text(encoding='utf8'))

def write(path, value):
    path = Path(path); temp = path.with_suffix(path.suffix+'.writing')
    temp.write_text(json.dumps(value, indent=2, allow_nan=False)+'\n', encoding='utf8')
    temp.replace(path)

def validate_junit(path):
    root = ET.parse(path).getroot(); suites = list(root.iter('testsuite'))
    if not suites or any(int(s.get(k,'0')) for s in suites for k in ('errors','failures','skipped')):
        raise ValueError('Require zero actual failures/errors/skips, including Linux symlink test')
    actual = []
    for case in root.iter('testcase'):
        if any(case.find(k) is not None for k in ('error','failure','skipped')):
            raise ValueError('Every actual testcase must pass')
        actual.append((case.get('classname'), case.get('name')))
    if len(actual) != 20 or len(set(actual)) != 20 or sorted(actual) != sorted(CASES):
        raise ValueError('Require exactly the frozen twenty distinct testcase identities')
    return dict(tests=20, passed=20, skipped=0, failures=0, errors=0, actual_testcases=actual)

def verify_source():
    path = P/'research_v2/incoming'/('segment_sources_'+COMMIT[:7]+'.json')
    if sha(path) != MANIFEST_SHA: raise ValueError('Frozen source manifest changed')
    manifest = read(path)
    if manifest['code_commit'] != COMMIT: raise ValueError('Source manifest commit differs')
    for name, expected in manifest['files'].items():
        candidate = SOURCE/name
        if (candidate.is_symlink() or not candidate.is_file() or candidate.stat().st_size != expected['bytes']
                or sha(candidate) != expected['sha256']):
            raise ValueError('Immutable archive source bytes differ: '+name)
    if sha(SOURCE/CONFIG) != CONFIG_SHA: raise ValueError('Registered diagnostic config changed')
    return dict(path=str(path), sha256=MANIFEST_SHA, files=len(manifest['files']),
                archive_sha256=manifest['archive_sha256'], config_sha256=CONFIG_SHA,
                all_archive_dependencies_verified=True)

def prior_tests():
    status = read(VALIDATION/'tests.status.json')
    receipt = read(VALIDATION/'validation_receipt.json')
    identity = read(VALIDATION/'validation_identity.json')
    test_wrapper = P/'research_v2/incoming'/('segment_tests_'+COMMIT[:7]+'.sh')
    if (status.get('status') != 'completed' or status.get('exit_code') != 0 or status.get('code_commit') != COMMIT
            or receipt.get('status') != 'passed' or receipt.get('code_commit') != COMMIT
            or receipt.get('identity_sha256') != sha(VALIDATION/'validation_identity.json')
            or receipt.get('pytest_xml_sha256') != sha(VALIDATION/'pytest.xml')
            or identity.get('stage') != 'tests' or identity.get('code_commit') != COMMIT
            or identity.get('source_manifest',{}).get('sha256') != MANIFEST_SHA
            or identity.get('runner',{}).get('sha256') != sha(__file__)
            or identity.get('wrapper',{}).get('path') != str(test_wrapper)
            or identity.get('wrapper',{}).get('sha256') != sha(test_wrapper)):
        raise ValueError('Actual twenty tests on this exact source/runner must pass before diagnostic')
    validate_junit(VALIDATION/'pytest.xml')
    return {name:sha(VALIDATION/name) for name in ('tests.status.json','validation_receipt.json',
                                                 'validation_identity.json','pytest.xml')}
